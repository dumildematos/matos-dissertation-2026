% --- Skeleton helpers ---
% Grey marker for text still to be written.
\newcommand{\towrite}[1][Text to write.]{{\color[gray]{.55}\textit{[#1]}}}

% \thesisimage[includegraphics options]{path}
% Shows a labelled box until the file at `path` is uploaded to the project.
\newcommand{\thesisimage}[2][width=\linewidth]{%
  \IfFileExists{#2}{\includegraphics[#1]{#2}}{%
    \fbox{\parbox[c][2.5cm][c]{.9\linewidth}{\centering\scriptsize\nolinkurl{#2}}}}}

% \thesisfigure[includegraphics options]{path}{caption}{label}
\newcommand{\thesisfigure}[4][width=.8\linewidth]{%
  \begin{figure}[!htbp]
    \centering
    \thesisimage[#1]{#2}
    \caption{#3}\label{#4}
  \end{figure}}

% \thesissubfigure{width}{path}{caption}{label}: use inside a figure environment.
\newcommand{\thesissubfigure}[4]{%
  \begin{subfigure}[t]{#1}
    \centering
    \thesisimage{#2}
    \caption{#3}\label{#4}
  \end{subfigure}}

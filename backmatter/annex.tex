% Appendices: supporting documents, listed in the index without page numbers.
\clearpage
\indexedsection{Appendices}
\addtocontents{toc}{\protect\cftpagenumbersoff{subsection}}

\indexedsubsection{Appendix A — Experimental Protocol}

\towrite[Aim: give the complete protocol. Include: step-by-step session script; participant instructions; task definitions and cue timing; session checklist; abort and emergency procedures.]

\indexedsubsection{Appendix B — Participant Information and Informed Consent Documents}

\towrite[Include only if applicable. Aim: provide the ethics documents. Include: participant information sheet; blank consent form; questionnaires; the ethics approval reference, if granted. Exclude anything that identifies a participant.]

\indexedsubsection{Appendix C — Supplementary Experimental Results}

\towrite[Aim: hold detail too long for Chapter 5. Include: per-participant and per-session tables; extra charts; full statistical outputs.]

\indexedsubsection{Appendix D — System Architecture and Technical Configuration}

\towrite[Aim: give the full technical record. Include: detailed diagrams; car wiring or hardware details; configuration parameters and thresholds; message format; software versions.]

\indexedsubsection{Appendix E — Source Code and Repository Information}

\towrite[Aim: point readers to the code. Include: repository URL; the release or commit used in the study; licence; folder structure; any key listings. Never include API credentials.]

\indexedsubsection{Appendix F — Instructions for Replicating the Experimental Data}

\towrite[Aim: let others repeat the analysis. Include: where the anonymised data is stored and how to access it; data dictionary; installation steps; commands that reproduce each result.]
